141@gym_bp.route("/gym")
142def gym_page():
143
144    if not session.get("username"):
145        return redirect(url_for("auth.login"))
146    ensure_default_gym_exercises()
147    # get the user who is logged in
148    user = get_logged_in_user()
149
150    if not user:
151        return redirect(url_for("auth.login"))
152    monday = get_week_start()
153
154    # all the exer sorted by muscle group then name
155    all_exercises = GymExercise.query.order_by(
156        GymExercise.muscle_group, GymExercise.name
157    ).all()
158
159    # list of different muscle groups
160    m_groups = (
161        db.session.query(GymExercise.muscle_group)
162        .distinct()
163        .order_by(GymExercise.muscle_group)
164        .all()
165    )
166    # list
167    m_groups = [g[0] for g in m_groups]
168
169    # get all the assignments the useer given by a trainer
170    rows1 = (
171        db.session.query(GymAssignment, GymExercise, User)
172        .join(GymExercise, GymAssignment.gym_exercise_id == GymExercise.gym_exercise_id)
173        .join(User, GymAssignment.trainer_id == User.user_id)
174        .filter(GymAssignment.client_id == user.user_id)
175        .order_by(GymAssignment.date_assigned.desc())
176        .all()
177    )
178
179    # put extra info onto each assignment so the template can show it
180    my_assignments = []
181    for a, ex, trainer in rows1:
182        a.exercise_name = ex.name
183        a.muscle_group = ex.muscle_group
184        a.trainer_name = f"{trainer.first_name} {trainer.last_name}"
185        my_assignments.append(a)
186
187    # the workouts user done
188    rows2 = (
189        db.session.query(GymWorkout, GymExercise)
190        .join(GymExercise, GymWorkout.gym_exercise_id == GymExercise.gym_exercise_id)
191        .filter(GymWorkout.user_id == user.user_id)
192        .order_by(GymWorkout.date.desc(), GymWorkout.gym_workout_id.desc())
193        .all()
194    )
195
196    # add the exers name/ muscle type to each workout
197    my_workouts = []
198    for w, ex in rows2:
199        w.exercise_name = ex.name
200        w.muscle_group = ex.muscle_group
201        my_workouts.append(w)
202
203    # count how many different days the user went to g
204    gym_days = (
205        db.session.query(func.count(func.distinct(GymWorkout.date)))
206        .filter(GymWorkout.user_id == user.user_id, GymWorkout.date >= monday)
207        .scalar()
208        or 0
209    )
210
211    # add all sets
212    sets_week = (
213        db.session.query(func.coalesce(func.sum(GymWorkout.sets_completed), 0))
214        .filter(GymWorkout.user_id == user.user_id, GymWorkout.date >= monday)
215        .scalar()
216        or 0
217    )
218
219    # muscle type user worked the most this week
220    best_muscle = (
221        db.session.query(
222            GymExercise.muscle_group, func.count(GymWorkout.gym_workout_id)
223        )
224        .join(GymWorkout, GymWorkout.gym_exercise_id == GymExercise.gym_exercise_id)
225        .filter(GymWorkout.user_id == user.user_id, GymWorkout.date >= monday)
226        .group_by(GymExercise.muscle_group)
227        .order_by(func.count(GymWorkout.gym_workout_id).desc())
228        .first()
229    )
230
231    # none if nthing
232    best_muscle_name = best_muscle[0] if best_muscle else "None yet"
233    clients = []
234
235    if user.role == "pt":
236        clients = (
237            User.query.join(SessionBooking, SessionBooking.client_id == User.user_id)
238            .filter(
239                SessionBooking.trainer_id == user.user_id,
240                SessionBooking.status == "confirmed",
241            )
242            .distinct()
243            .order_by(User.username.asc())
244            .all()
245        )
246
247    # send everything to the html page
248    return render_template(
249        "gym.html",
250        user=user,
251        clients=clients,
252        exercises=all_exercises,
253        muscle_groups=m_groups,
254        assignments=my_assignments,
255        workouts=my_workouts,
256        sessions_this_week=gym_days,
257        total_sets_week=sets_week,
258        top_muscle_name=best_muscle_name,
259    )

306def assign_exercise():
307    # the trainer
308    trainer = get_logged_in_user()
309
310    if not trainer:
311        return redirect(url_for("auth.login"))
312
313    if trainer.role != "pt":
314        flash("Only trainers can assign exercises.", "error")
315        return redirect(url_for("gym.gym_page"))
316
317    # get the form data
318    c_id = request.form.get("client_id", type=int)
319    ex_id = request.form.get("gym_exercise_id", type=int)
320    client = User.query.filter_by(user_id=c_id, role="customer").first()
321
322    if not client:
323        flash("Client was not found.", "error")
324        return redirect(url_for("gym.gym_page"))
325
326    num_sets = request.form.get("sets", type=int)
327    num_reps = request.form.get("reps", type=int)
328    kg = request.form.get("weight_kg", type=float)
329    note = request.form.get("notes")
330
331    # they have to choose one clint or error
332    if not c_id or not ex_id:
333        flash("Client and exercise are required.", "error")
334        return redirect(url_for("gym.gym_page"))
335
336    # make assignment
337    new_assign = GymAssignment(
338        trainer_id=trainer.user_id,
339        client_id=c_id,
340        gym_exercise_id=ex_id,
341        sets=num_sets or 0,
342        reps=num_reps or 0,
343        weight_kg=kg or 0.0,
344        notes=note or "",
345        date_assigned=date.today(),
346    )
347
348    db.session.add(new_assign)
349    db.session.commit()
350
351    flash("Exercise assigned!", "success")
352    return redirect(url_for("gym.gym_page"))
